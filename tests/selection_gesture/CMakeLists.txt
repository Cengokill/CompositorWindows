cmake_minimum_required(VERSION 3.31.6)
project(SelectionGestureTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(selection_gesture_tests SelectionGestureTests.cpp ../../src/editing/SelectionGesture.cpp ../../src/editing/Selection.cpp ../../src/core/Document.cpp ../../src/graphics/upstream/WandPixels.c)
target_include_directories(selection_gesture_tests PRIVATE ../../src)
target_compile_definitions(selection_gesture_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(selection_gesture_tests PRIVATE /W4 /WX /utf-8)
target_link_libraries(selection_gesture_tests PRIVATE d2d1 windowscodecs ole32)
enable_testing()
foreach(case polygon_corners click_history marquee_rounding centered_programmatic ellipse option_marquee fresh_shift stationary_modifiers polygon_closure quarter_pixel mode_and_interrupt empty_noop move_axis autoscroll geometry_equality)
 add_test(NAME selection_gesture.${case} COMMAND selection_gesture_tests ${case})
 set_tests_properties(selection_gesture.${case} PROPERTIES TIMEOUT 20)
endforeach()
