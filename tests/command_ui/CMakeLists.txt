cmake_minimum_required(VERSION 3.31.6)
project(CommandUiTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Gui Widgets Test)
add_executable(command_ui_tests CommandRegistryTests.cpp ../../src/ui/CommandRegistry.cpp)
target_include_directories(command_ui_tests PRIVATE ../../src)
target_compile_options(command_ui_tests PRIVATE /W4 /WX /utf-8)
target_link_libraries(command_ui_tests PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Test)
enable_testing()
foreach(case source_predicates layer_eligibility pending_preparation stale_activation text_shortcuts text_menu_routes catalog_manifest registration preparation_error)
 add_test(NAME command_ui.${case} COMMAND command_ui_tests ${case})
 set_tests_properties(command_ui.${case} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
endforeach()
