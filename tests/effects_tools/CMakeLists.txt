cmake_minimum_required(VERSION 3.31.6)
project(CompositorEffectsToolsTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Gui Widgets Test Concurrent)
add_executable(effects_tools_tests effects_tools_tests.cpp ../../src/effects_tools/Levels.cpp ../../src/effects_tools/Hue.cpp ../../src/effects_tools/Color.cpp ../../src/effects_tools/Curves.cpp ../../src/graphics/StackRenderer.cpp ../../src/ui/AdjustmentAdvancedControls.cpp ../../src/core/Document.cpp ../../src/graphics/upstream/LevelsPixels.c)
target_include_directories(effects_tools_tests PRIVATE ../../src)
target_compile_definitions(effects_tools_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(effects_tools_tests PRIVATE /W4 /utf-8)
target_link_libraries(effects_tools_tests PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Test ole32)
enable_testing()
foreach(case levels_normalization_handles levels_histogram_weights levels_histogram_display levels_auto_modes levels_sample_calibration levels_original_sampler levels_json_preservation levels_widget_controls hue_band_weights hue_sample_range hue_target_drag hue_json_spectrum curves_graph_drag curves_json color_hex_hsb color_sample_ring hue_widget_controls curves_widget_controls)

 set_tests_properties(effects_tools.${case} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endforeach()

file(GLOB UPSTREAM_C CONFIGURE_DEPENDS ../../src/graphics/upstream/*.c)
add_executable(adjustment_dialog_tests adjustment_dialog_tests.cpp ../../src/ui/AdjustmentDialog.cpp ../../src/ui/AdjustmentAdvancedControls.cpp ../../src/effects_tools/Levels.cpp ../../src/effects_tools/Hue.cpp ../../src/effects_tools/Color.cpp ../../src/effects_tools/Curves.cpp ../../src/graphics/StackRenderer.cpp ../../src/render/SoftwareRenderer.cpp ../../src/effects/Adjustments.cpp ../../src/editing/Selection.cpp ../../src/layers/LayerOperations.cpp ../../src/core/Document.cpp ../../src/graphics/PixelAlgorithms.cpp ${UPSTREAM_C})

target_compile_options(adjustment_dialog_tests PRIVATE /W4 /utf-8)
target_link_libraries(adjustment_dialog_tests PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Test Qt6::Concurrent d2d1 windowscodecs ole32)

endforeach()
