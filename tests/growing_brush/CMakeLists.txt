cmake_minimum_required(VERSION 3.31.6)
project(GrowingBrushTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(growing_brush_tests ../graphics/GrowingBrushSessionTests.cpp ../../src/graphics/GrowingBrushSession.cpp ../../src/graphics/BrushSession.cpp ../../src/graphics/BrushCoverage.cpp ../../src/graphics/D3D11BrushCoverage.cpp ../../src/graphics/StackRenderer.cpp ../../src/core/Document.cpp)
target_include_directories(growing_brush_tests PRIVATE ../../src)
target_compile_definitions(growing_brush_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(growing_brush_tests PRIVATE /W4 /WX /utf-8)
target_link_libraries(growing_brush_tests PRIVATE ole32 d3d11 d3dcompiler dxgi)
enable_testing()
foreach(case unchanged_grid blank_30000 growth_padding_masks transformed_growth selection_cancel_mask atomic_budget immutable_history empty_erase_noop gpu_equivalence invalid_sample viewport_image viewport_transformed viewport_mask region_row_copies small_source_growth)
 add_test(NAME growing_brush.${case} COMMAND growing_brush_tests ${case})
endforeach()
set_tests_properties(growing_brush.small_source_growth PROPERTIES TIMEOUT 10)
add_executable(signed_division_probe signed_division_probe.cpp)
target_compile_options(signed_division_probe PRIVATE /O2 /W4 /WX /utf-8)
