cmake_minimum_required(VERSION 3.31.6)
project(CompositorCacheTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core)
file(GLOB UPSTREAM_C CONFIGURE_DEPENDS ../../src/graphics/upstream/*.c)
add_executable(core_cache_tests ../core_cache_tests.cpp ../../src/core/Document.cpp ../../src/core/CompositeCache.cpp ../../src/graphics/StackRenderer.cpp ../../src/render/SoftwareRenderer.cpp ../../src/effects/Adjustments.cpp ../../src/graphics/PixelAlgorithms.cpp ${UPSTREAM_C})
target_include_directories(core_cache_tests PRIVATE ../../src)
target_compile_definitions(core_cache_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _USE_MATH_DEFINES)
target_compile_options(core_cache_tests PRIVATE /W4 /WX /utf-8)
target_link_libraries(core_cache_tests PRIVATE Qt6::Core ole32)
add_executable(stack_renderer_tests ../graphics/StackRendererTests.cpp ../../src/graphics/StackRenderer.cpp ../../src/core/Document.cpp)
target_include_directories(stack_renderer_tests PRIVATE ../../src)
target_compile_definitions(stack_renderer_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(stack_renderer_tests PRIVATE /W4 /WX /utf-8)
target_link_libraries(stack_renderer_tests PRIVATE ole32)
enable_testing()
foreach(case blank_30000 identity_fastpath normal_equivalence offscreen_invalidation bounded_eviction metadata_and_hidden_source full_after_partial lod_large_content scaled_renderer_contract scaled_grain_partition outside_and_rejections blank_culling lod_offscreen_invalidation unsupported_blank_adjustment)
 add_test(NAME cache.${case} COMMAND core_cache_tests ${case})
endforeach()
add_test(NAME cache.stack_renderer_regression COMMAND stack_renderer_tests)
