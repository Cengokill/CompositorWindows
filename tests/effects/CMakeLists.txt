set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core)
file(GLOB UPSTREAM_C CONFIGURE_DEPENDS ../../src/graphics/upstream/*.c)
add_executable(effects_tests effects_tests.cpp ../../src/effects/Adjustments.cpp ../../src/graphics/PixelAlgorithms.cpp ../../src/core/Document.cpp ${UPSTREAM_C})

target_compile_definitions(effects_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _USE_MATH_DEFINES)
target_compile_options(effects_tests PRIVATE /W4 /utf-8)
target_link_libraries(effects_tests PRIVATE Qt6::Core ole32)
enable_testing()
