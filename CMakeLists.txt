# Windows searches System32 before PATH for imported DLLs. Keep the locked
# codecs and ORT runtime beside every development executable, as packaging does.
file(READ "${CMAKE_SOURCE_DIR}/dependencies/imaging/lock.json" imaging_lock)
string(JSON imaging_binary_count LENGTH "${imaging_lock}" binaries)
math(EXPR imaging_binary_last "${imaging_binary_count}-1")
set(imaging_runtime_files)
foreach(index RANGE ${imaging_binary_last})
  string(JSON binary_path GET "${imaging_lock}" binaries ${index} path)
  string(JSON expected_hash GET "${imaging_lock}" binaries ${index} sha256)
  set(binary_file "${CMAKE_SOURCE_DIR}/dependencies/imaging/${binary_path}")
  file(SHA256 "${binary_file}" actual_hash)
  if(NOT actual_hash STREQUAL expected_hash)
    message(FATAL_ERROR "Locked imaging runtime checksum mismatch: ${binary_path}")
  endif()
  list(APPEND imaging_runtime_files "${binary_file}")
endforeach()
